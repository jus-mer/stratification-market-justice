# Manual ESEC construction (9 classes + retired + unemployed)

    isco1 = floor(isco88 / 1000),  # 1-digit major group

    # ESEC extended with labor force status
    esec_ext = case_when(
      # Primero: categorías fuera de la fuerza de trabajo (m02)
      m02 == 3 ~ 10,  # Jubilado/pensionado
      m02 == 7 ~ 11,  # Desempleado
      
      # Luego: ESEC 9-class para quienes tienen ocupación
      # Class 1: Higher salariat (managers, professionals with employees/supervisory)
      selfemp_esec == FALSE & is_supervisor == TRUE & isco1 %in% c(1, 2) ~ 1,
      selfemp_esec == TRUE & n_employees >= 10 & isco1 %in% c(1, 2) ~ 1,
      
      # Class 2: Lower salariat (professionals, managers non-supervisory)
      selfemp_esec == FALSE & is_supervisor == FALSE & isco1 %in% c(1, 2) ~ 2,
      selfemp_esec == FALSE & isco1 == 3 & isco2 %in% c(31, 32, 34) ~ 2,
      
      # Class 3: Higher white collar (technicians, associate professionals supervisory)
      selfemp_esec == FALSE & is_supervisor == TRUE & isco1 == 3 ~ 3,
      selfemp_esec == FALSE & is_supervisor == TRUE & isco1 == 4 ~ 3,
      
      # Class 4: Lower white collar (clerks, service workers non-supervisory)
      selfemp_esec == FALSE & is_supervisor == FALSE & isco1 == 3 ~ 4,
      selfemp_esec == FALSE & is_supervisor == FALSE & isco1 == 4 ~ 4,
      
      # Class 5: Higher blue collar (skilled craft workers supervisory)
      selfemp_esec == FALSE & is_supervisor == TRUE & isco1 %in% c(6, 7, 8) ~ 5,
      
      # Class 6: Skilled workers (craft workers non-supervisory)

      # Class 9: Unskilled workers (elementary occupations)
